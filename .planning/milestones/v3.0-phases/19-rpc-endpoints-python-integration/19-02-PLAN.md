```python
def write_signal(self, name: str, symbol: str, timestamp: datetime, attributes: dict = None) -> str:
    """Write a global TradeSignal via WriteSignal RPC (per D-01: no playground_id).

    Args:
        name: Signal name (e.g., 'ma_crossover', 'covered_call')
        symbol: Stock symbol (e.g., 'AAPL')
        timestamp: Signal timestamp
        attributes: Optional key-value attributes (values converted to strings)

    Returns:
        signal_id: UUID string assigned by the server
    """
    ts = Timestamp()
    ts.FromDatetime(timestamp)

    attrs = {}
    if attributes:
        attrs = {k: str(v) for k, v in attributes.items()}

    request = WriteSignalRequest(
        name=name,
        symbol=symbol,
        timestamp=ts,
        attributes=attrs,
    )
    try:
        response = self.network_call_with_retry('write_signal', self.client.WriteSignal, request)
    except Exception as e:
        self.logger.exception("failed to connect to gRPC service (write_signal)")
        raise e
    return response.signal_id
```

```python
def get_processed_signals(self, name: str = None, symbol: str = None,
                          from_timestamp: datetime = None, to_timestamp: datetime = None) -> list:
    """Get signals consumed by this playground (per D-03/RPC-01).

    Args:
        name: Optional filter by signal name
        symbol: Optional filter by symbol
        from_timestamp: Optional start of time range
        to_timestamp: Optional end of time range

    Returns:
        List of TradeSignalProto objects consumed by this playground
    """
    kwargs = {"playground_id": self.id}
    if name is not None:
        kwargs["name"] = name
    if symbol is not None:
        kwargs["symbol"] = symbol
    if from_timestamp is not None:
        ts = Timestamp()
        ts.FromDatetime(from_timestamp)
        kwargs["from_timestamp"] = ts
    if to_timestamp is not None:
        ts = Timestamp()
        ts.FromDatetime(to_timestamp)
        kwargs["to_timestamp"] = ts

    request = GetProcessedSignalsRequest(**kwargs)
    try:
        response = self.network_call_with_retry('get_processed_signals', self.client.GetProcessedSignals, request)
    except Exception as e:
        self.logger.exception("failed to connect to gRPC service (get_processed_signals)")
        raise e
    return response.signals
```

Note: `self.id` is the playground_id string already stored on the client instance.

```python
"""
Datasource scripts for TradeSignal production.

Each datasource module produces TradeSignals for a specific signal type.
Datasources follow a dual-mode pattern:

  - **Sim mode**: Strategy imports the module directly and calls produce_signals()
    to generate signals in-process (no RPC needed).

    Example:
        from datasources.ma_crossover import produce_signals
        signals = produce_signals(candles, params)

  - **Live mode**: Run the module from __main__ which calls WriteSignal RPC
    to send signals to the Go server's global signal store.

    Example:
        python -m datasources.ma_crossover --symbol AAPL --server http://localhost:5051

See base.py for the skeleton pattern that all datasource modules should follow.
"""
```
