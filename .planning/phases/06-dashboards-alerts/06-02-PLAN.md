```yaml
# Grafana unified alerting provisioning
# Ref: https://grafana.com/docs/grafana/latest/alerting/set-up/provision-alerting-resources/file-provisioning/
apiVersion: 1

# Contact point: Slack webhook for alert notifications (per D-06)
contactPoints:
  - orgId: 1
    name: grodt-slack
    receivers:
      - uid: grodt-slack-webhook
        type: slack
        settings:
          url: ${SLACK_OPTION_ALERTS_WEBHOOK_URL}
          title: |
            [{{ .Status | toUpper }}] {{ .CommonLabels.alertname }}
          text: |
            {{ range .Alerts }}
            **{{ .Labels.alertname }}** - {{ .Annotations.summary }}
            {{ .Annotations.description }}
            {{ end }}

# Notification policy: route critical alerts to Slack + Grafana UI
policies:
  - orgId: 1
    receiver: grafana-default-email
    group_by:
      - grafana_folder
      - alertname
    group_wait: 30s
    group_interval: 5m
    repeat_interval: 4h
    routes:
      - receiver: grodt-slack
        object_matchers:
          - ["severity", "=", "critical"]
        continue: false

# Alert rule groups
groups:
  # Heartbeat staleness rules (ALERT-01, ALERT-02)
  - orgId: 1
    name: grodt-heartbeat
    folder: Grodt Alerts
    interval: 30s
    rules:
      # ALERT-01: Go server heartbeat stale (>2 minutes, per D-04)
      - uid: go-heartbeat-stale
        title: Go Server Heartbeat Stale
        condition: C
        for: 2m
        annotations:
          summary: Go server heartbeat has not been received for over 2 minutes
          description: The grodt-server has stopped emitting heartbeat metrics. Check if the server process is running.
        labels:
          severity: critical
        data:
          - refId: A
            relativeTimeRange:
              from: 300
              to: 0
            datasourceUid: prometheus
            model:
              expr: absent_over_time(grodt_heartbeat_uptime_seconds[2m])
              instant: true
              refId: A
          - refId: C
            relativeTimeRange:
              from: 300
              to: 0
            datasourceUid: __expr__
            model:
              type: threshold
              expression: A
              conditions:
                - evaluator:
                    type: gt
                    params:
                      - 0
                  operator:
                    type: and
                  reducer:
                    type: last
              refId: C

      # ALERT-02: Python strategy heartbeat stale (>2 minutes)
      - uid: python-heartbeat-stale
        title: Python Strategy Heartbeat Stale
        condition: C
        for: 2m
        annotations:
          summary: Python strategy heartbeat has not been received for over 2 minutes
          description: The grodt-strategy has stopped emitting heartbeat metrics. Check if the Python strategy process is running.
        labels:
          severity: critical
        data:
          - refId: A
            relativeTimeRange:
              from: 300
              to: 0
            datasourceUid: prometheus
            model:
              expr: absent_over_time(grodt_strategy_heartbeat[2m])
              instant: true
              refId: A
          - refId: C
            relativeTimeRange:
              from: 300
              to: 0
            datasourceUid: __expr__
            model:
              type: threshold
              expression: A
              conditions:
                - evaluator:
                    type: gt
                    params:
                      - 0
                  operator:
                    type: and
                  reducer:
                    type: last
              refId: C

  # Error rate rules (ALERT-03)
  - orgId: 1
    name: grodt-errors
    folder: Grodt Alerts
    interval: 1m
    rules:
      # ALERT-03: Error log rate exceeds 5 errors per minute (Claude's discretion on threshold per D-05)
      - uid: error-rate-spike
        title: Error Rate Spike
        condition: C
        for: 5m
        annotations:
          summary: Error log rate exceeds 5 errors per minute
          description: Grodt services are producing errors at an elevated rate. Check Loki logs for details.
        labels:
          severity: critical
        data:
          - refId: A
            relativeTimeRange:
              from: 600
              to: 0
            datasourceUid: loki
            model:
              expr: 'sum(rate({service_name=~"grodt.*"} | logfmt | level=`error` [5m]))'
              instant: true
              refId: A
          - refId: C
            relativeTimeRange:
              from: 600
              to: 0
            datasourceUid: __expr__
            model:
              type: threshold
              expression: A
              conditions:
                - evaluator:
                    type: gt
                    params:
                      - 0.083
                  operator:
                    type: and
                  reducer:
                    type: last
              refId: C
```

**Key design decisions:**
- Heartbeat alerts use `absent_over_time(...[2m])` which returns 1 when no samples exist in the window (per D-04 default threshold)
- `for: 2m` pending period prevents spurious alerts during cold start (Pitfall 5 from research)
- Error rate threshold: 0.083 = 5 errors per 60 seconds (Claude's discretion per D-05)
- Error rate `for: 5m` to avoid transient spikes
- Slack contact point uses `${SLACK_OPTION_ALERTS_WEBHOOK_URL}` env var substitution (per D-06)
- All alerts severity=critical to route through Slack notification policy
